#蒙特卡罗投点绘制心形
```
import matplotlib.pyplot as plt
import random

# 解决中文与负号显示问题
plt.rcParams['font.sans-serif'] = ['Microsoft YaHei', 'SimHei']
plt.rcParams['axes.unicode_minus'] = False

# 采样区域范围（完全包含心形）
x_min, x_max = -1.8, 1.8
y_min, y_max = -1.5, 1.8
n_samples = 20000  # 投点总数，越多图形越清晰

# 存储心形内部的点
heart_x, heart_y = [], []

for _ in range(n_samples):
    x = random.uniform(x_min, x_max)
    y = random.uniform(y_min, y_max)
    # 心形隐函数判定：满足不等式则点在心形内部
    if (x**2 + y**2 - 1) ** 3 - x**2 * y**3 <= 0:
        heart_x.append(x)
        heart_y.append(y)

# 绘图
plt.figure(figsize=(6, 6))
plt.scatter(heart_x, heart_y, c='crimson', s=2)
plt.title("蒙特卡罗投点绘制心形")
plt.axis('equal')
plt.axis('off')  # 隐藏坐标轴，更像纯图形
plt.show()

```

#字符田字格绘制
```
def draw_grid(n):  #n为每个格子的宽度
    line="+"+"-"*n
    row="❘"+" "*n
    for i in range(n):
        print(line*n+"+")
        for j in range(n):
            print(row*n+"|")
    print(line*n+"+")

n=int(input("请输入田字格宽度:"))
draw_grid(n)
```
